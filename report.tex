\section{Tính năng 5: Tiền xử lý và Chuẩn hóa dữ liệu}
Bộ dữ liệu sau khi đọc vào gồm $10,000$ mẫu khách hàng với $14$ thuộc tính ban đầu. Trước khi đưa vào mạng ANN, dữ liệu phải trải qua quy trình tiền xử lý nghiêm ngặt bao gồm loại bỏ các cột không liên quan (\texttt{RowNumber}, \texttt{CustomerId}, \texttt{Surname}), mã hóa các cột phân loại dạng chữ (\texttt{Geography}, \texttt{Gender}) bằng \texttt{LabelEncoder}, One-Hot Encoding tránh bẫy đa cộng tuyến, phân chia tập Train/Test theo tỷ lệ 80/20 phân tầng và áp dụng chuẩn hóa đặc trưng Z-score:

\begin{equation}
    z = \frac{x - \mu}{\sigma}
\end{equation}

\begin{lstlisting}[language=Python, caption=Mã nguồn tiền xử lý dữ liệu và chuẩn hóa đặc trưng]
# 6. Tien xu ly va Chuan hoa du lieu
# Tach bien doc lap (X) va bien phu thuoc (y)
X = dataset.drop(labels=['CustomerId', 'Surname', 'RowNumber', 'Exited'], axis=1)
y = dataset['Exited']

# Ma hoa Label Encoding cho Geography va Gender
label_geo = LabelEncoder()
X['Geography'] = label_geo.fit_transform(X['Geography'])

label_gender = LabelEncoder()
X['Gender'] = label_gender.fit_transform(X['Gender'])

# Tranh bay da cong tuyen bang cach ap dung One-Hot Encoding cho Geography
X = pd.get_dummies(X, drop_first=True, columns=['Geography'])

# Phan chia tap du lieu thanh Train (80%) va Test (20%)
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=0, stratify=y)

# Chuan hoa dac trung bang StandardScaler
scaler = StandardScaler()
X_train = scaler.fit_transform(X_train)
X_test = scaler.transform(X_test)
\end{lstlisting}

\section{Tính năng 6: Xây dựng mô hình mạng nơ-ron nhân tạo (ANN)}
Mô hình mạng nơ-ron nhân tạo được thiết kế theo cấu trúc tuần tự (\texttt{Sequential}) liên kết toàn phần (\texttt{Dense}). Mạng ANN được xây dựng gồm 3 tầng liên kết:
\begin{enumerate}
    \item \textbf{Tầng ẩn thứ nhất (Input \& Hidden Layer 1):} Nhận $11$ đặc trưng đầu vào, gồm $11$ neuron liên kết toàn phần, sử dụng hàm kích hoạt phi tuyến \texttt{ReLU}: $f(x) = \max(0, x)$.
    \item \textbf{Tầng ẩn thứ hai (Hidden Layer 2):} Gồm $128$ neuron liên kết toàn phần với hàm kích hoạt \texttt{ReLU}, cho phép mạng học các đặc trưng phi tuyến sâu.
    \item \textbf{Tầng đầu ra (Output Layer):} Gồm $1$ neuron duy nhất với hàm kích hoạt \texttt{Sigmoid} để đưa ra xác suất rời bỏ dịch vụ $p \in [0, 1]$: $S(x) = \frac{1}{1 + e^{-x}}$.
\end{enumerate}
Mô hình được biên dịch bằng bộ tối ưu hóa thích nghi \texttt{Adam} và hàm mất mát Entropy chéo nhị phân (\texttt{Binary Cross-Entropy}):

\begin{lstlisting}[language=Python, caption=Mã nguồn xây dựng cấu trúc mạng ANN]
# 7. Xay dung cau truc mang ANN tuan tu
model = Sequential()
# Input layer (11 features) va Hidden layer 1 (11 nodes), activation ReLU
model.add(Dense(X.shape[1], activation='relu', input_dim=X.shape[1]))
# Hidden layer 2 (128 nodes), activation ReLU
model.add(Dense(128, activation='relu'))
# Output layer (1 node), activation Sigmoid
model.add(Dense(1, activation='sigmoid'))

# Bien dich mo hinh mang ANN
model.compile(optimizer='adam', loss='binary_crossentropy', metrics=['accuracy'])
model.summary()
\end{lstlisting}

\section{Tính năng 7: Đào tạo mô hình (Train/Fit ANN)}
Quá trình đào tạo mô hình được thiết lập thực thi qua $10$ epochs với kích thước phân nhóm dữ liệu (batch size) bằng $10$. Đồng thời, $10\%$ tập huấn luyện được tách ra làm tập xác thực (\texttt{validation\_split=0.1}) nhằm mục đích theo dõi trực tiếp quá trình hội tụ của mô hình:

\begin{lstlisting}[language=Python, caption=Huấn luyện mô hình ANN]
# 8. Dao tao mo hinh ANN
history = model.fit(
    X_train, 
    y_train.to_numpy(), 
    batch_size=10, 
    epochs=10, 
    validation_split=0.1, 
    verbose=1
)
\end{lstlisting}

\section{Tính năng 8: Kiểm tra và Đánh giá hiệu năng mô hình}
Đánh giá mô hình trên tập kiểm thử độc lập gồm $2,000$ khách hàng hoàn toàn mới để đo lường các giá trị thực tế của Test Loss và Test Accuracy. Tiếp theo, dự đoán nhãn lớp phân loại (với ngưỡng quyết định xác suất $0.5$) để in ra Ma trận nhầm lẫn (Confusion Matrix) cùng Báo cáo phân loại chi tiết (Classification Report):

\begin{lstlisting}[language=Python, caption=Đánh giá hiệu năng và kiểm thử mô hình]
# 9. Kiem tra va Danh gia mo hinh tren tap Test doc lap
test_loss, test_acc = model.evaluate(X_test, y_test.to_numpy(), verbose=0)
print(f"Test Loss: {test_loss:.4f}")
print(f"Test Accuracy: {test_acc * 100:.2f}%")

# Du doan ket qua nhan lop tren tap Test
y_pred = model.predict(X_test)
y_pred = (y_pred > 0.5).astype("int32")

# Tinh toan ma tran nham len va in classification report
cm = confusion_matrix(y_test, y_pred)
print("Confusion Matrix:\n", cm)
print("\nClassification Report:\n", classification_report(y_test, y_pred))
\end{lstlisting}

\section{Tính năng 9: Sử dụng mô hình để dự đoán thực tế}
Ứng dụng mô hình đã đào tạo để đưa ra quyết định phân loại dự đoán cho một khách hàng mới cụ thể với các thông số: Pháp, Điểm tín dụng 600, Nam, 40 tuổi, gắn bó 3 năm, số dư 60,000 USD, sử dụng 2 sản phẩm, có thẻ tín dụng, là thành viên tích cực và mức lương ước lượng 50,000 USD. 

Đoạn mã này thực hiện tiền xử lý (One-Hot Encoder cho quốc gia, chuẩn hóa vector đầu vào bằng `scaler.transform`) và đưa ra xác suất dự đoán cùng kết luận cuối cùng:

\begin{lstlisting}[language=Python, caption=Dự báo khả năng rời đi cho khách hàng mới]
# 10. Su dung mo hinh de du doan mot khach hang moi cu the
# CreditScore: 600, Gender: Male (1), Age: 40, Tenure: 3, Balance: 60000, 
# NumOfProducts: 2, HasCrCard: 1, IsActiveMember: 1, EstimatedSalary: 50000
# Geography: France (Geography_1 = 0, Geography_2 = 0)
new_cust = [[600, 1, 40, 3, 60000, 2, 1, 1, 50000, 0, 0]]

# Bat buoc phai chuan hoa du lieu dau vao bang scaler da fit tu truoc
new_cust_scaled = scaler.transform(new_cust)

# Dua vao mang ANN du doan xac suat va nhan lop
prob = model.predict(new_cust_scaled)
churn_decision = (prob > 0.5).astype("int32")

print("Xac suat roi di cua khach hang:", prob[0][0])
if churn_decision[0][0] == 1:
    print("Ket luan: Khach hang se ROI BO dich vu (CHURN)")
else:
    print("Ket luan: Khach hang se O LAI (STAY)")
\end{lstlisting}

Kết quả xác suất dự đoán của khách hàng này cực kỳ thấp ($\approx 3.32\% < 50\%$). Do đó, mô hình đưa ra kết quả dự đoán nhãn lớp là $0$, khẳng định khách hàng này sẽ **Ở LẠI** sử dụng dịch vụ của ngân hàng.
